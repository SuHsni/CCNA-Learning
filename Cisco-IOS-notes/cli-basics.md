# first mode => user exec mode :

switch>

router>

# second mode => privileged mode :

switch#

router#

# global mode :

switch(config)#

router(config)#

# list of commands or help :

?

# turn on privileged mode :

enable or en

# return to user mode :

disable or disa or exit

# auto complete :

tab

# turn on global config :

configure terminal or conf t

# when the text gets split across the terminal

ctrl + r

# return from any of the above modes to privileged mode

ctrl + z or end

# delete the entire line

ctrl + u

# delete word by word

ctrl + w

# go to the beginning of the line

ctrl + a

# go to the end of the line

ctrl + e
